cmake_minimum_required(VERSION 3.16)
project(rm_dynamic_obstacle_critic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep nav2_mppi_controller nav2_costmap_2d nav2_msgs pluginlib rclcpp rclcpp_lifecycle tf2_ros tf2 geometry_msgs nav_msgs diagnostic_msgs visualization_msgs rm_competition_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
set(deps nav2_mppi_controller nav2_costmap_2d nav2_msgs pluginlib rclcpp rclcpp_lifecycle tf2_ros tf2 geometry_msgs nav_msgs diagnostic_msgs visualization_msgs rm_competition_interfaces)
add_library(${PROJECT_NAME} SHARED src/dynamic_obstacle_critic.cpp src/static_stopping_critic.cpp src/native_cycle_snapshot_critic.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} ${deps})
pluginlib_export_plugin_description_file(nav2_mppi_controller critics.xml)
add_executable(dynamic_safety_guard src/safety_guard.cpp)
target_include_directories(dynamic_safety_guard PRIVATE include)
ament_target_dependencies(dynamic_safety_guard ${deps})
add_executable(native_command_observer src/native_command_observer.cpp)
target_include_directories(native_command_observer PRIVATE include)
ament_target_dependencies(native_command_observer rclcpp geometry_msgs)
install(TARGETS native_command_observer RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} dynamic_safety_guard ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_model test/test_model.cpp)
  target_include_directories(test_model PRIVATE include)
  ament_target_dependencies(test_model ${deps})
  ament_add_gtest(test_plugin test/test_plugin.cpp)
  target_include_directories(test_plugin PRIVATE include)
  ament_target_dependencies(test_plugin ${deps})
  target_link_libraries(test_plugin ${PROJECT_NAME})
  # Offline only; deliberately does not link this package's plugin library, so
  # the same executable can load frozen/new libraries through ament/pluginlib.
  add_executable(benchmark_stopping_plugin tools/benchmark_stopping_plugin.cpp)
  ament_target_dependencies(benchmark_stopping_plugin ${deps})
  add_executable(replay_native_optimizer tools/replay_native_optimizer.cpp)
  ament_target_dependencies(replay_native_optimizer nav2_mppi_controller rclcpp)
  # Match the locally pinned Humble 1.1.20 SDK's inlined SG arithmetic. These
  # options apply only to this offline tool, never our runtime plugins/guard.
  include(CheckCXXCompilerFlag)
  function(configure_pinned_native_sdk target)
    target_compile_definitions(${target} PRIVATE XTENSOR_ENABLE_XSIMD XTENSOR_USE_XSIMD)
    foreach(option -mno-avx512f -msse4.2 -mavx2 -mfma
        -O3 -finline-limit=10000000 -ffp-contract=fast -ffast-math -mtune=generic)
      string(MAKE_C_IDENTIFIER "RM_NATIVE_REPLAY_${option}" flag_id)
      check_cxx_compiler_flag("${option}" ${flag_id})
      if(${flag_id})
        target_compile_options(${target} PRIVATE "${option}")
      endif()
    endforeach()
  endfunction()
  configure_pinned_native_sdk(replay_native_optimizer)
  add_executable(native_safe_control_witness tools/native_safe_control_witness.cpp)
  ament_target_dependencies(native_safe_control_witness nav2_mppi_controller rclcpp)
  configure_pinned_native_sdk(native_safe_control_witness)
endif()
ament_export_include_directories(include)
ament_export_dependencies(${deps})
ament_package()
