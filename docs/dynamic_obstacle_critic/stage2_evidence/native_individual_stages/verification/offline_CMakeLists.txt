cmake_minimum_required(VERSION 3.16)
project(rm_native_individual_offline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(rclcpp REQUIRED)
foreach(target native_safe_control_witness native_individual_stages)
  add_executable(${target} /repo/src/rm_dynamic_obstacle_critic/tools/native_safe_control_witness.cpp)
  ament_target_dependencies(${target} nav2_mppi_controller rclcpp)
  target_compile_definitions(${target} PRIVATE XTENSOR_ENABLE_XSIMD XTENSOR_USE_XSIMD)
  target_compile_options(${target} PRIVATE -mno-avx512f -msse4.2 -mavx2 -mfma -O3 -finline-limit=10000000 -ffp-contract=fast -ffast-math -mtune=generic)
endforeach()
target_compile_definitions(native_individual_stages PRIVATE RM_NATIVE_INDIVIDUAL_STAGES=1)
