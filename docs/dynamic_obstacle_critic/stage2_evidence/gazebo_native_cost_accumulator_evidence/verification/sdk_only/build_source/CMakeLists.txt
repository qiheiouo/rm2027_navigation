cmake_minimum_required(VERSION 3.16)
project(rm_native_cost_accumulator_offline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_gtest REQUIRED)
foreach(dep nav2_mppi_controller nav2_costmap_2d pluginlib rclcpp rclcpp_lifecycle)
  find_package(${dep} REQUIRED)
endforeach()
add_library(rm_native_cost_accumulator_observer SHARED /repo/src/rm_dynamic_obstacle_critic/src/native_cost_accumulator_observer.cpp)
target_include_directories(rm_native_cost_accumulator_observer PRIVATE /repo/src/rm_dynamic_obstacle_critic/include)
ament_target_dependencies(rm_native_cost_accumulator_observer nav2_mppi_controller nav2_costmap_2d pluginlib rclcpp rclcpp_lifecycle)
ament_add_gtest(test_cost_accumulator_evidence /repo/src/rm_dynamic_obstacle_critic/test/test_cost_accumulator_evidence.cpp SKIP_LINKING_MAIN_LIBRARIES)
target_include_directories(test_cost_accumulator_evidence PRIVATE /repo/src/rm_dynamic_obstacle_critic/include)
ament_target_dependencies(test_cost_accumulator_evidence nav2_mppi_controller)
target_link_libraries(test_cost_accumulator_evidence gtest)
pluginlib_export_plugin_description_file(nav2_mppi_controller cost_accumulator_critics.xml)
install(TARGETS rm_native_cost_accumulator_observer LIBRARY DESTINATION lib)
ament_add_gtest(test_cost_accumulator_plugin /repo/src/rm_dynamic_obstacle_critic/test/test_cost_accumulator_plugin.cpp SKIP_LINKING_MAIN_LIBRARIES)
ament_target_dependencies(test_cost_accumulator_plugin nav2_mppi_controller pluginlib rclcpp rclcpp_lifecycle)
target_link_libraries(test_cost_accumulator_plugin gtest)
ament_package()
