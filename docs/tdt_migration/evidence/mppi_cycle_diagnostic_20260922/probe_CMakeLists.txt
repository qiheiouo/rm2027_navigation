cmake_minimum_required(VERSION 3.16)
project(tdt_trace_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp nav2_costmap_2d nav2_core tf2_geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(trace_probe trace_probe.cpp cycle_trace.cpp)
target_include_directories(trace_probe PRIVATE include /opt/ros/humble/include)
ament_target_dependencies(trace_probe rclcpp nav2_costmap_2d nav2_core tf2_geometry_msgs)
target_compile_options(trace_probe PRIVATE -g -O1 -fno-omit-frame-pointer -fsanitize=address,undefined -Wall -Wextra -Werror)
target_link_options(trace_probe PRIVATE -fsanitize=address,undefined)
