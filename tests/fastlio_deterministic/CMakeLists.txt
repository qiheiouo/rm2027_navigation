cmake_minimum_required(VERSION 3.16)
project(fastlio_deterministic_validation LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp sensor_msgs nav_msgs geometry_msgs tf2 tf2_ros tf2_eigen tf2_geometry_msgs pcl_conversions livox_ros_driver2)
  find_package(${dep} REQUIRED)
endforeach()
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(OpenMP REQUIRED)
foreach(fixture imu measurement)
  add_executable(${fixture}_fixture ${fixture}_fixture.cpp)
  target_compile_features(${fixture}_fixture PRIVATE cxx_std_17)
  target_compile_options(${fixture}_fixture PRIVATE -O0 -g0 -D_GLIBCXX_ASSERTIONS)
  target_compile_definitions(${fixture}_fixture PRIVATE FASTLIO_SOURCE="${FASTLIO_SOURCE}" ROOT_DIR="${FASTLIO_SOURCE}/")
  target_include_directories(${fixture}_fixture PRIVATE ${FASTLIO_SOURCE} ${FASTLIO_SOURCE}/include ${PCL_INCLUDE_DIRS})
  target_link_libraries(${fixture}_fixture ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
  ament_target_dependencies(${fixture}_fixture rclcpp sensor_msgs nav_msgs geometry_msgs tf2 tf2_ros tf2_eigen tf2_geometry_msgs pcl_conversions livox_ros_driver2)
endforeach()
target_link_libraries(measurement_fixture ${FASTLIO_LIBRARY})
target_compile_definitions(measurement_fixture PRIVATE FASTLIO_REFERENCE="${FASTLIO_REFERENCE}")

# Optional local closure trace; existing fixtures and assertions remain intact.

  target_compile_features(bundle_state_trace PRIVATE cxx_std_17)
  target_compile_options(bundle_state_trace PRIVATE -O0 -g0 -D_GLIBCXX_ASSERTIONS)
  target_compile_definitions(bundle_state_trace PRIVATE BUNDLE_VISIBLE="${BUNDLE_VISIBLE}" ROOT_DIR="${FASTLIO_SOURCE}/")
  target_include_directories(bundle_state_trace PRIVATE ${FASTLIO_SOURCE} ${FASTLIO_SOURCE}/src ${FASTLIO_SOURCE}/include ${PCL_INCLUDE_DIRS})
  target_link_libraries(bundle_state_trace ${FASTLIO_LIBRARY} ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
  ament_target_dependencies(bundle_state_trace rclcpp sensor_msgs nav_msgs geometry_msgs tf2 tf2_ros tf2_eigen tf2_geometry_msgs pcl_conversions livox_ros_driver2)
endif()
