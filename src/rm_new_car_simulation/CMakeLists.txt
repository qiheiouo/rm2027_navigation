cmake_minimum_required(VERSION 3.16)
project(rm_new_car_simulation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
add_executable(s2_tf_owner_probe src/s2_tf_owner_probe.cpp)
target_compile_features(s2_tf_owner_probe PRIVATE cxx_std_17)
ament_target_dependencies(s2_tf_owner_probe rclcpp tf2_msgs)
install(TARGETS s2_tf_owner_probe DESTINATION lib/${PROJECT_NAME})
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(rm_omni_wheel_drive SHARED src/omni_wheel_drive.cpp)
target_compile_features(rm_omni_wheel_drive PUBLIC cxx_std_17)
target_compile_options(rm_omni_wheel_drive PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rm_omni_wheel_drive gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS rm_omni_wheel_drive LIBRARY DESTINATION lib)
install(DIRECTORY config assets launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/baseline_scene.py scripts/simulation_tf.py scripts/validate_baseline.py
  scripts/smoke_baseline.py
  DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_test REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  ament_add_test(s2_geometry COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test/test_s2_geometry.py" TIMEOUT 60 GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
endif()
ament_package()
