cmake_minimum_required(VERSION 3.16)
project(rm_r4_nav2_controller LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
option(RM_R4_BUILD_NAV2_CONTROLLER "Build the optional typed-cycle R4 Nav2 plugin" OFF)
if(RM_R4_BUILD_NAV2_CONTROLLER)
  find_package(nav2_core REQUIRED)
  find_package(pluginlib REQUIRED)
  find_package(rm_r4_prediction_consumption REQUIRED)
  find_package(rm_navigation_execution_adapters REQUIRED)
  if(NOT TARGET rm_r4_prediction_consumption::rm_r4_prediction_consumption_follow)
    message(FATAL_ERROR "Explicitly build A08 RM_R4_BUILD_FOLLOW with its fixed OSQP prefix first")
  endif()
  add_library(${PROJECT_NAME} SHARED src/controller.cpp src/proposal_bridge.cpp)
  target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
  target_include_directories(${PROJECT_NAME} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(${PROJECT_NAME} PUBLIC
    rm_r4_prediction_consumption::rm_r4_prediction_consumption_follow
    rm_navigation_execution_adapters::rm_navigation_execution_adapters)
  ament_target_dependencies(${PROJECT_NAME} PUBLIC nav2_core pluginlib)
  target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
  pluginlib_export_plugin_description_file(nav2_core plugin.xml)
  if(BUILD_TESTING)
    find_package(ament_cmake_gtest REQUIRED)
    ament_add_gtest(test_controller test/test_controller.cpp TIMEOUT 30)
    target_link_libraries(test_controller ${PROJECT_NAME})
    target_include_directories(test_controller PRIVATE ../rm_r4_prediction_consumption/test)
  endif()
  install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME}
    LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
  install(DIRECTORY include/ DESTINATION include)
  ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
  ament_export_dependencies(nav2_core pluginlib rm_r4_prediction_consumption rm_navigation_execution_adapters)
endif()
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
