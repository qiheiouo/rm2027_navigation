cmake_minimum_required(VERSION 3.16)
project(rm_stvl_temporal_research)
find_package(ament_cmake REQUIRED)
foreach(p rclcpp nav2_core nav2_costmap_2d nav2_mppi_controller pluginlib tf2_ros tf2 std_msgs rm_competition_interfaces)
 find_package(${p} REQUIRED)
endforeach()
find_package(nlohmann_json REQUIRED)
add_library(study_plugins SHARED plugins.cpp)
target_compile_features(study_plugins PUBLIC cxx_std_17)
ament_target_dependencies(study_plugins rclcpp nav2_core nav2_costmap_2d nav2_mppi_controller pluginlib tf2_ros tf2 std_msgs rm_competition_interfaces)
target_link_libraries(study_plugins nlohmann_json::nlohmann_json)
pluginlib_export_plugin_description_file(nav2_core controller.xml)
pluginlib_export_plugin_description_file(nav2_mppi_controller critics.xml)
install(TARGETS study_plugins LIBRARY DESTINATION lib)
add_executable(temporal_check check.cpp)
ament_target_dependencies(temporal_check rclcpp nav2_mppi_controller nav2_costmap_2d pluginlib std_msgs)
target_link_libraries(temporal_check nlohmann_json::nlohmann_json)
ament_package()
