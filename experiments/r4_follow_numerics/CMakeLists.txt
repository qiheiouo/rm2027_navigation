cmake_minimum_required(VERSION 3.16)
project(r4_follow_numerics LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rm_r4_prediction_consumption REQUIRED)
find_package(rm_navigation_execution_adapters REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(osqp REQUIRED CONFIG)
add_executable(endpoint endpoint.cpp instrumented.cpp)
if(NOT EXISTS "${R4_BASELINE_SOURCE}")
  message(FATAL_ERROR "Generate follow.cpp from baseline 0374f4b3; set R4_BASELINE_SOURCE")
endif()
target_compile_definitions(endpoint PRIVATE R4_NUMERICS_BASELINE_SOURCE="${R4_BASELINE_SOURCE}")
target_compile_features(endpoint PRIVATE cxx_std_20)
target_link_libraries(endpoint PRIVATE rm_r4_prediction_consumption::rm_r4_prediction_consumption
  rm_navigation_execution_adapters::rm_navigation_execution_adapters Eigen3::Eigen OpenSSL::Crypto osqp::osqp)
target_compile_options(endpoint PRIVATE -Wall -Wextra -Wpedantic)
add_executable(feedback endpoint.cpp plain.cpp)
target_compile_features(feedback PRIVATE cxx_std_20)
target_link_libraries(feedback PRIVATE rm_r4_prediction_consumption::rm_r4_prediction_consumption_follow
  rm_navigation_execution_adapters::rm_navigation_execution_adapters)
target_compile_options(feedback PRIVATE -Wall -Wextra -Wpedantic)
