cmake_minimum_required(VERSION 3.16)
project(r4_world_xy LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rm_r4_prediction_consumption REQUIRED)
foreach(target probe replay fixed_probe)
  if(target STREQUAL "fixed_probe")
    add_executable(${target} ../../src/rm_r4_prediction_consumption/test/follow_probe.cpp)
  else()
    add_executable(${target} ${target}.cpp)
  endif()
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(${target} rm_r4_prediction_consumption::rm_r4_prediction_consumption_follow
    rm_navigation_execution_adapters::rm_navigation_execution_adapters)
endforeach()
ament_target_dependencies(replay rclcpp rosbag2_cpp)
